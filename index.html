const f = localStorage.getItem('jan_favs'); if (f) setFavorites(JSON.parse(f));

const r = localStorage.getItem('jan_recent'); if (r) setRecentSearches(JSON.parse(r));

const v = localStorage.getItem('jan_viewed'); if (v) setRecentlyViewed(JSON.parse(v));

const rm = localStorage.getItem('jan_reminders'); if (rm) setReminders(JSON.parse(rm));

}, []);

const raw = MEDICINES_RAW_DATA.data;

const list = raw.slice(1).map(item => ({

drug_code: String(item.A || ''),

drug_name: String(item.B || ''),

unit_size: String(item.C || ''),

mrp: String(item.D || 'N/A'),

category: String(item.E || 'General')

const cats = ['All', ...Array.from(new Set(list.map(m => m.category))).sort()];

let res = medicinesList;

if (activeCategory !== 'All') res = res.filter(m => m.category === activeCategory);

if (searchTerm) {

const t = searchTerm.toLowerCase();

res = res.filter(m => m.drug_name.toLowerCase().includes(t) || m.drug_code.toLowerCase().includes(t));

}

return res;

}, [searchTerm, activeCategory, medicinesList]);

e.stopPropagation();

localStorage.setItem('jan_favs', JSON.stringify(next));

};

localStorage.setItem('jan_viewed', JSON.stringify(next));

};

const next = [...reminders, { id: Date.now(), drug_name: med.drug_name, time, active: true }];
